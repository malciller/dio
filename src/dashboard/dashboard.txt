  |> List.mapi (fun i l -> if i land 1 = 0 then I.string Style.header l else I.string Style.header_ac l)

let fmt_runtime start =
  let secs = int_of_float (Unix.gettimeofday () -. start) in
  Printf.sprintf "⟨ %02dh:%02dm:%02ds ⟩" (secs / 3600) (secs mod 3600 / 60) (secs mod 60)

let bar ?(max_len = 20) n = I.hcat (List.init (min max_len n) (fun _ -> power))

let get_term_dimensions () = match Notty_unix.winsize Unix.stdout with | Some (h,w) -> (max 24 h, max 80 w) | None -> (24,80)

(* price ladder *)
let price_ladder current_price buy_orders sell_orders =
  let buys  = List.sort (fun (p1,_) (p2,_) -> compare p2 p1) buy_orders in
  let sells = List.sort (fun (p1,_) (p2,_) -> compare p1 p2) sell_orders in
  let _,tw  = get_term_dimensions () in
  let cw     = tw - 4 in
  let lw     = cw - 22 in
  let ladder = Array.make lw (I.string A.empty " ") in
  let prices = current_price :: List.map fst (buys @ sells) in
  let min_p  = List.fold_left min max_float prices in
  let max_p  = List.fold_left max min_float prices in
  let range  = max 0.0001 (max_p -. min_p) in
  let idx p  = max 0 (min (lw-1) (int_of_float ((p -. min_p) /. range *. float_of_int (lw-1)))) in
  List.iter (fun (p,_) -> ladder.(idx p) <- buy_arrow) buys;
  List.iter (fun (p,_) -> ladder.(idx p) <- sell_arrow) sells;
  ladder.(idx current_price) <- price_marker;
  I.hcat (Array.to_list ladder)

(* row *)
let format_price asset price = match Kraken.Ws_feed.get_price_precision asset with | Some prec -> Printf.sprintf "%.*f" prec price | None -> Printf.sprintf "%.2f" price

let row_of_asset asset =
  let open I in
  let cp_opt = Stats.get_price asset in
  let buys,sells = Stats.get_orders_for_symbol asset in
  let get_closest lst cmp = match cp_opt with | None -> None | Some cp -> let c = Float.of_string (Primitives.Price.to_string cp) in List.fold_left (fun acc (p,_) -> if cmp p c then match acc with None->Some p | Some best -> Some (if cmp best p then best else p) else acc) None lst in
  let closest_buy  = get_closest buys (fun p c -> p <= c) in
  let closest_sell = get_closest sells (fun p c -> p >= c) in
  let price_s = function None -> "-.--" | Some p -> format_price asset p in
  let stats = hcat [ I.string Style.asset (Printf.sprintf "%-7s" asset); I.string Style.border " │ "; I.string Style.buy "B:"; I.string Style.buy (price_s closest_buy); I.string Style.border " │ "; I.string Style.now "P:"; I.string Style.now (match cp_opt with None->"-.--" | Some p->format_price asset (Float.of_string (Primitives.Price.to_string p))); I.string Style.border " │ "; I.string Style.sell "S:"; I.string Style.sell (price_s closest_sell); I.string Style.border " │ "; I.string Style.orders (Printf.sprintf "◎%2d" (List.length sells)) ] in
  let ladder_img = match cp_opt with Some p -> price_ladder (Float.of_string (Primitives.Price.to_string p)) buys sells | None -> I.string Style.sell "no-price" in
  let _,tw = get_term_dimensions () in
  let cw = tw - 4 in
  let row = stats <|> void 2 1 <|> ladder_img in
  hcat [I.string Style.border "┃ "; row; void (cw - width row - 2) 1; I.string Style.border "┃"]

let rec take n = function | [] -> [] | _ when n<=0 -> [] | x::xs -> x :: take (n-1) xs

let priority_assets = ["BTC/USD";"SOL/USD";"ETH/USD";"ADA/USD";"TRX/USD"]
let asset_rank a = match List.mapi (fun i t -> (t,i)) priority_assets |> List.assoc_opt a with | Some r -> r | None -> List.length priority_assets
let compare_assets a b = compare (asset_rank a) (asset_rank b)

type state = { mutable show_logs: bool; mutable frame: int }
let make_state () = { show_logs=true; frame=0 }

let render st =
  let open I in
  let th,tw = get_term_dimensions () in
  let cw = tw-4 in
  let runtime = I.string Style.runtime ("Runtime: " ^ fmt_runtime Stats.start_ts) |> I.pad ~t:1 ~l:2 in
  let assets = M.fold (fun k _ acc -> k::acc) !Stats.pending_orders [] |> List.sort compare_assets |> List.map row_of_asset in
  let header_h,height_runtime,rows_h = height header, height runtime, List.fold_left (fun a r->a+height r) 0 assets in
  let logs_h = max 0 (th - (header_h + height_runtime + rows_h + 4)) in
  let main = vcat (header :: runtime :: assets) in
  let main_adj = if width main < cw then hcat [main; void (cw - width main) (height main)] else main in
  let logs_section = if not st.show_logs || logs_h < 3 then void 0 0 else let hdr = hcat [I.string Style.log_hdr "┏━ LOGS "; I.string Style.border (String.make (cw - 8) '-'); I.string Style.border "┓"] |> I.pad ~l:2 in let wrap msg = let rec aux acc s = if String.length s <= cw then List.rev (s::acc) else aux (String.sub s 0 cw :: acc) (String.sub s cw (String.length s - cw)) in aux [] msg in let log_lines = !Stats.dashboard_logs |> List.rev |> take (logs_h-1) |> List.concat_map wrap in let body = vcat (List.map (fun l->I.string Style.log_txt l) log_lines) in let img = vcat [hdr; body] in if width img < cw then hcat [img; void (cw - width img) (height img)] else img in
  vcat [void 0 1; main_adj; void 0 1; logs_section; void 0 1]

let start ~on_quit () : Notty_lwt.Term.t =
  let term = Notty_lwt.Term.create ~mouse:false () in
  let state = make_state () in
  let rec tick () = state.frame <- state.frame + 1; Notty_lwt.Term.image term (render state) >>= fun () -> Lwt.pause () >>= fun () -> Lwt_unix.sleep 1.0 >>= tick in
  let  handle_event = function | `Key ((`ASCII ('l'|'L')), _) -> state.show_logs <- not state.show_logs; Lwt.return_unit | `Key ((`ASCII ('q'|'Q')), _) -> on_quit () | _ -> Lwt.return_unit in
  let rec input_loop () = Lwt.catch (fun () -> Lwt_stream.get (Notty_lwt.Term.events term) >>= function Some ev -> handle_event ev >>= input_loop | None -> on_quit ()) (fun ex -> Lwt_log_core.error ~section:(Lwt_log_core.Section.make "dashboard") (Printf.sprintf "Input loop error: %s" (Printexc.to_string ex)) >>= fun () -> on_quit ()) in
  Lwt.async tick; Lwt.async input_loop; term
